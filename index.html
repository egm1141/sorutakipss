-- 1. Dersler tablosu
CREATE TABLE dersler (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  ad TEXT NOT NULL,
  renk TEXT DEFAULT '#6366f1',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Günlük kayıtlar tablosu
CREATE TABLE gunluk_kayitlar (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  tarih DATE NOT NULL DEFAULT CURRENT_DATE,
  ders_id UUID REFERENCES dersler(id) ON DELETE CASCADE,
  soru_sayisi INTEGER NOT NULL DEFAULT 0,
  not_metni TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. Row Level Security (public erişim)
ALTER TABLE dersler ENABLE ROW LEVEL SECURITY;
ALTER TABLE gunluk_kayitlar ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public dersler" ON dersler FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Public kayitlar" ON gunluk_kayitlar FOR ALL USING (true) WITH CHECK (true);
